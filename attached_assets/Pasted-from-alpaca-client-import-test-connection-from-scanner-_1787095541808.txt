from alpaca_client import (
    test_connection,
)

from scanner import (
    scan_market,
)

from strategy import (
    calculate_score,
)

from risk import (
    calculate_trade,
)

from config import (
    PAPER_TRADING,
    POSITION_SIZE,
    MIN_REWARD_RISK,
)


def print_signal(
    stock,
    strategy,
    trade
):

    print()
    print(
        "======================================"
    )

    print(
        f"🚨 TRADE CANDIDATE: "
        f"{stock['symbol']}"
    )

    print(
        "======================================"
    )

    print(
        f"Price:        "
        f"${stock['price']:.2f}"
    )

    print(
        f"Day gain:     "
        f"{stock['gain']:.2f}%"
    )

    print(
        f"Relative vol: "
        f"{stock['relative_volume']:.2f}x"
    )

    print(
        f"Strategy score:"
        f" {strategy['score']}"
    )

    print()

    print("CONFIRMATIONS:")

    for reason in strategy[
        "reasons"
    ]:

        print(
            f"  ✅ {reason}"
        )

    print()

    print("TRADE PLAN:")

    print(
        f"Shares:       "
        f"{trade['shares']}"
    )

    print(
        f"Entry:        "
        f"${trade['entry']:.2f}"
    )

    print(
        f"Stop:         "
        f"${trade['stop']:.2f}"
    )

    print(
        f"Target:       "
        f"${trade['target']:.2f}"
    )

    print(
        f"Risk:         "
        f"${trade['total_risk']:.2f}"
    )

    print(
        f"Potential:    "
        f"${trade['potential_profit']:.2f}"
    )

    print(
        "======================================"
    )


def main():

    print(
        """
==========================================
       MOMENTUM TRADING BOT
              VERSION 2
==========================================
"""
    )

    # --------------------------------------
    # HARD SAFETY CHECK
    # --------------------------------------

    if PAPER_TRADING is not True:

        raise RuntimeError(
            "LIVE TRADING IS DISABLED."
        )

    print(
        "MODE: PAPER TRADING"
    )

    print(
        f"MANUAL SHARES: "
        f"{POSITION_SIZE}"
    )

    print(
        f"MIN R/R: "
        f"{MIN_REWARD_RISK}:1"
    )

    # --------------------------------------
    # ALPACA
    # --------------------------------------

    test_connection()

    # --------------------------------------
    # SCAN
    # --------------------------------------

    candidates = scan_market()

    if not candidates:

        print(
            "No candidates found."
        )

        return

    # --------------------------------------
    # STRATEGY
    # --------------------------------------

    for stock in candidates:

        strategy = calculate_score(
            stock["data"]
        )

        print(
            f"\n{stock['symbol']} "
            f"Score = "
            f"{strategy['score']}"
        )

        if not strategy[
            "qualified"
        ]:

            print(
                "❌ Does not qualify"
            )

            continue

        # ----------------------------------
        # TRADE PLAN
        # ----------------------------------

        entry = stock["price"]

        trade = calculate_trade(

            stock["data"],

            entry

        )

        if trade is None:

            print(
                "❌ Invalid risk setup"
            )

            continue

        print_signal(
            stock,
            strategy,
            trade
        )

    print(
        """
==========================================
NO ORDERS WERE SENT.

VERSION 2 IS PAPER-ANALYSIS ONLY.

==========================================
"""
    )


if __name__ == "__main__":

    main()
