from datetime import datetime, timedelta
from zoneinfo import ZoneInfo

import pandas as pd

from alpaca.data.requests import StockBarsRequest
from alpaca.data.timeframe import TimeFrame

from alpaca.data.enums import DataFeed

from alpaca_client import stock_data_client

from config import DATA_FEED


ET = ZoneInfo("America/New_York")


def get_feed():

    if DATA_FEED.lower() == "sip":

        return DataFeed.SIP

    return DataFeed.IEX


def get_intraday_bars(
    symbol,
    minutes=1,
    lookback_minutes=240
):

    now = datetime.now(ET)

    start = now - timedelta(
        minutes=lookback_minutes
    )

    request = StockBarsRequest(

        symbol_or_symbols=symbol,

        timeframe=TimeFrame.Minute,

        start=start,

        end=now,

        feed=get_feed(),

    )

    response = stock_data_client.get_stock_bars(
        request
    )

    df = response.df

    if df.empty:

        return None

    if isinstance(df.index, pd.MultiIndex):

        df = df.xs(
            symbol,
            level="symbol"
        )

    df = df.copy()

    df.index = pd.to_datetime(
        df.index
    )

    df.sort_index(
        inplace=True
    )

    return df


def calculate_vwap(df):

    df = df.copy()

    typical_price = (
        df["high"]
        + df["low"]
        + df["close"]
    ) / 3

    cumulative_volume = (
        df["volume"].cumsum()
    )

    cumulative_pv = (
        typical_price
        * df["volume"]
    ).cumsum()

    df["vwap"] = (
        cumulative_pv
        / cumulative_volume
    )

    return df


def calculate_indicators(df):

    df = df.copy()

    df = calculate_vwap(df)

    df["ema9"] = (
        df["close"]
        .ewm(span=9)
        .mean()
    )

    df["ema20"] = (
        df["close"]
        .ewm(span=20)
        .mean()
    )

    df["volume_avg20"] = (
        df["volume"]
        .rolling(20)
        .mean()
    )

    df["relative_volume"] = (
        df["volume"]
        / df["volume_avg20"]
    )

    df["high_20"] = (
        df["high"]
        .rolling(20)
        .max()
        .shift(1)
    )

    return df
