from flask import Flask, render_template, request, jsonify
import requests
import datetime
import pytz

app = Flask(__name__)

# ------------------------------
# CONFIGURATION
# ------------------------------
ALPACA_BASE = "https://paper-api.alpaca.markets"
ALPACA_DATA = "https://data.alpaca.markets/v2"

# IMPORTANT: replace these with your real keys inside Replit Secrets
API_KEY = ""
API_SECRET = ""
HEADERS = {
    "APCA-API-KEY-ID": API_KEY,
    "APCA-API-SECRET-KEY": API_SECRET
}

# ------------------------------
# GET ASSET LIST
# ------------------------------
def get_assets():
    url = f"{ALPACA_BASE}/v2/assets"
    r = requests.get(url, headers=HEADERS)
    if r.status_code == 200:
        return [a["symbol"] for a in r.json() if a["tradable"]]
    return []

# ------------------------------
# ORB STRATEGY LOGIC
# ------------------------------
def orb_trade(symbol, take_profit):

    now = datetime.datetime.now(pytz.timezone("America/New_York"))
    market_open = now.replace(hour=9, minute=45, second=0, microsecond=0)  # 15-min ORB

    if now < market_open:
        return {"status": "WAIT", "message": "ORB window has not started yet."}

    # Get 15m candle (first 15 min after open)
    data_url = f"{ALPACA_DATA}/stocks/{symbol}/bars?timeframe=15Min&limit=1"
    r = requests.get(data_url, headers=HEADERS)

    if r.status_code != 200:
        return {"status": "ERROR", "message": "Error loading market data."}

    bar = r.json()["bars"][0]
    high = bar["h"]
    low = bar["l"]

    # Get current price
    last_url = f"{ALPACA_DATA}/stocks/{symbol}/trades/latest"
    lr = requests.get(last_url, headers=HEADERS)
    price = lr.json()["trade"]["p"]

    # Determine breakout
    if price > high:
        direction = "BUY"
    elif price < low:
        direction = "SELL"
    else:
        return {"status": "NO_BREAKOUT", "message": "No breakout yet."}

    # Submit trade
    order_url = f"{ALPACA_BASE}/v2/orders"
    qty = 1

    order = {
        "symbol": symbol,
        "qty": qty,
        "side": direction.lower(),
        "type": "market",
        "time_in_force": "day"
    }

    orr = requests.post(order_url, json=order, headers=HEADERS)

    if orr.status_code != 200:
        return {"status": "ORDER_ERROR", "message": "Trade failed."}

    # Submit take-profit order
    tp_price = round(price * (1 + take_profit), 2)

    tp_order = {
        "symbol": symbol,
        "qty": qty,
        "side": "sell" if direction == "BUY" else "buy",
        "type": "limit",
        "limit_price": tp_price,
        "time_in_force": "day"
    }

    requests.post(order_url, json=tp_order, headers=HEADERS)

    return {
        "status": "TRADE_PLACED",
        "direction": direction,
        "entry_price": price,
        "take_profit": tp_price
    }

# ------------------------------
# ROUTES
# ------------------------------
@app.route("/")
def index():
    return render_template("index.html", assets=get_assets())

@app.route("/run_orb", methods=["POST"])
def run_orb():
    symbol = request.json["symbol"]
    take_profit = float(request.json["take_profit"])
    return jsonify(orb_trade(symbol, take_profit))

# ------------------------------
# START SERVER
# ------------------------------
app.run(host="0.0.0.0", port=8080)
