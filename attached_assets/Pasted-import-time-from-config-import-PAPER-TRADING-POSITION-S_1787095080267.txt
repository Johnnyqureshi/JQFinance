import time

from config import (
    PAPER_TRADING,
    POSITION_SIZE,
    MAX_TRADES_PER_DAY,
    MAX_DAILY_LOSS,
)

from alpaca_client import (
    test_connection,
)

from scanner import scan_market

from strategy import evaluate_stock

from risk import (
    calculate_trade,
    print_trade_plan,
)


def main():

    print("""
========================================
       MOMENTUM TRADING BOT
========================================
        VERSION 1
        PAPER TRADING
========================================
""")

    # ---------------------------------
    # SAFETY CHECK
    # ---------------------------------

    if PAPER_TRADING is not True:

        raise RuntimeError(
            "SAFETY ERROR: Version 1 only supports paper trading."
        )

    print("Trading mode: PAPER")
    print(f"Manual position size: {POSITION_SIZE} shares")
    print(f"Maximum trades/day: {MAX_TRADES_PER_DAY}")
    print(f"Maximum daily loss: ${MAX_DAILY_LOSS:.2f}")

    # ---------------------------------
    # TEST ALPACA CONNECTION
    # ---------------------------------

    test_connection()

    # ---------------------------------
    # SCAN
    # ---------------------------------

    stocks = scan_market()

    if not stocks:

        print("No stocks passed the initial scanner.")
        return

    # ---------------------------------
    # STRATEGY EVALUATION
    # ---------------------------------

    qualified = []

    print("\n========== STRATEGY RESULTS ==========")

    for stock in stocks:

        result = evaluate_stock(stock)

        print(
            f"{result['symbol']} | "
            f"Score: {result['score']} | "
            f"Qualified: {result['qualified']}"
        )

        if result["reasons"]:

            for reason in result["reasons"]:

                print(f"   + {reason}")

        if result["qualified"]:

            qualified.append(result)

    print("=======================================\n")

    # ---------------------------------
    # TRADE PLANS
    # ---------------------------------

    for stock in qualified:

        trade = calculate_trade(
            stock["price"]
        )

        print_trade_plan(
            stock["symbol"],
            trade
        )

    # ---------------------------------
    # IMPORTANT
    # ---------------------------------
    #
    # NO ORDERS ARE SENT IN VERSION 1.
    #
    # We first verify that the scanner,
    # strategy and risk engine behave
    # correctly.
    # ---------------------------------

    print("""
========================================
VERSION 1 COMPLETE

NO ORDERS WERE SENT.

Next stage:
- Intraday data
- News/catalyst detection
- Float filtering
- VWAP
- High-of-day detection
- Bull flag detection
- Level 2
- Actual paper-order execution
========================================
""")


if __name__ == "__main__":
    main()
